    function renderDriverDashboard_V7() {         const h = dispatchDrivers.map(d => {             const plate = d.plate;             // ⚡️ V8.2: 動態扣重邏輯 (排除已完成/送達/結案)             const currentLoad = dispatchAllTasks.filter(t => {                 const isMatched = (String(t.assignedDriver).trim() === d.name.trim() || String(t.vehicle).trim() === plate.trim());                 const isFinished = (t.status && (t.status.includes('完成') || t.status.includes('送達') || t.status.includes('結案')));                 return isMatched && !isFinished;             }).reduce((sum, tk) => sum + (Number(tk.weight) || 0), 0);              const maxWeight = (plate.includes('BX') || plate.includes('BND')) ? 6000 : 3500;             const pct = Math.min(100, (currentLoad / maxWeight) * 100);             const barColor = pct >= 90 ? 'var(--color-danger)' : pct >= 70 ? 'var(--color-accent)' : 'var(--color-success)';             return `                 <div class="driver-weight-card" onclick="handleDriverClick('${escapeHTML(plate)}','${escapeHTML(d.name)}')">                     <div style="display:flex; justify-content:space-between; align-items:center;">                         <b style="color:#fff; font-size:15px;">${escapeHTML(d.name)}</b>                         <span style="font-size:10px; color:var(--color-primary); font-family:'Rajdhani';">${escapeHTML(plate)}</span>                     </div>                     <div class="load-bar-bg"><div class="load-bar-fill" style="width:${pct}%; background:${barColor};"></div></div>                     <div style="display:flex; justify-content:space-between; margin-top:5px; font-weight:900; font-size:13px;">                         <span style="color:${barColor};">${currentLoad.toFixed(0)} <span style="font-size:10px;">kg</span></span>                         <span style="color:#94a3b8;">${pct.toFixed(0)}%</span>                     </div>                 </div>`;         }).join('');         document.getElementById('driverLoadDashboard').innerHTML = h || '<div style="color:#475569; padding:20px; text-align:center;">載入中...</div>';
